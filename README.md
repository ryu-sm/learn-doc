import pyautogui
import time
from datetime import datetime

# 每隔多少秒移动一次
INTERVAL = 60

# 鼠标移动距离（像素）
MOVE_DISTANCE = 20

# 鼠标移动时间（秒）
MOVE_DURATION = 0.5

pyautogui.FAILSAFE = True

def simulate_mouse():
    x, y = pyautogui.position()

    screen_width, screen_height = pyautogui.size()

    # 避免移动到屏幕边缘
    if x + MOVE_DISTANCE < screen_width:
        target_x = x + MOVE_DISTANCE
    else:
        target_x = x - MOVE_DISTANCE

    # 向右或向左移动
    pyautogui.moveTo(
        target_x,
        y,
        duration=MOVE_DURATION
    )

    # 移动回原位置
    pyautogui.moveTo(
        x,
        y,
        duration=MOVE_DURATION
    )

def main():
    print("鼠标自动移动脚本已启动")
    print("按 Ctrl+C 停止运行")

    try:
        while True:
            simulate_mouse()

            print(
                datetime.now().strftime("%H:%M:%S"),
                "鼠标移动完成"
            )

            time.sleep(INTERVAL)

    except KeyboardInterrupt:
        print("脚本已停止")

if __name__ == "__main__":
    main()






import ctypes
import time
from datetime import datetime

# Windows 电源管理标志
ES_CONTINUOUS = 0x80000000
ES_SYSTEM_REQUIRED = 0x00000001
ES_DISPLAY_REQUIRED = 0x00000002

kernel32 = ctypes.windll.kernel32

def keep_awake():
    result = kernel32.SetThreadExecutionState(
        ES_CONTINUOUS |
        ES_SYSTEM_REQUIRED |
        ES_DISPLAY_REQUIRED
    )

    if result == 0:
        raise ctypes.WinError()

def restore_power_settings():
    kernel32.SetThreadExecutionState(
        ES_CONTINUOUS
    )

if __name__ == "__main__":
    try:
        keep_awake()
        print("Windows 防休眠模式已启动")

        while True:
            print(
                datetime.now().strftime("%H:%M:%S"),
                "电脑保持唤醒状态"
            )
            time.sleep(60)

    except KeyboardInterrupt:
        print("正在退出...")

    finally:
        restore_power_settings()
        print("已恢复 Windows 默认电源管理")

# 000_新人入場の駆け込み寺

## 1. 概要

本ドキュメントは、新規入場者が業務開始に必要な端末、アカウントおよびネットワーク環境を設定するための手順をまとめたものです。

各設定作業は、関連する手順書を参照し、記載された順序に従って実施してください。

---

## 2. 端末の受領

入場時に、以下の端末および付属品を受領してください。

| No. | 項目 | 付属品・関連資料 | 備考 |
|---|---|---|---|
| 1 | デスクトップPC | 電源ケーブル、「ご使用の前に…」、PC設定手順書 | |
| 2 | モニター | 電源ケーブル、ディスプレイケーブル | |
| 3 | ノートPC | ACアダプター、「パートナーシンクラの利用について」、リモート手順書 | |

### 確認事項

- 受領した端末および付属品に不足がないこと。
- 電源ケーブルおよびディスプレイケーブルが正常に接続できること。
- デスクトップPCおよびモニターが正常に起動すること。
- ノートPCの電源が正常に入ること。
- 必要な設定手順書を参照できること。
